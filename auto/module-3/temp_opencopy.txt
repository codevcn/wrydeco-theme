        function openCopyModal(e, type = 'id') {
            if(e) e.stopPropagation();
            switchCopyTab(type);
            copyModal.classList.remove('hidden');
            copyCountInput.value = '';
            setTimeout(() => copyCountInput.focus(), 50);
        }

        function closeCopyModal() {
            copyModal.classList.add('hidden');
            copyCountInput.blur();
        }
